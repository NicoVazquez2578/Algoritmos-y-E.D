// Devuelve el árbol de expansión mínimo de "grafo", empezando por el vértice "origen".
// Si el grafo no es conexo, devuelve el árbol de la componente del origen.
IUndirectedGraph<V, D> prim(IUndirectedGraph<V, D> grafo, Comparable<V> origen)
COMIENZO
    inicio ← grafo.buscarVertice(origen)
    SI inicio = nulo ENTONCES
        DEVOLVER nulo                                   // el origen no está en el grafo
    FIN SI

    U ← nuevo HashSet<V>()                              // vértices ya en el árbol
    U.agregar(inicio)
    VmenosU ← nuevo HashSet<V>(grafo.vertices())        // vértices que faltan
    VmenosU.remover(inicio)

    arbol ← nuevo UndirectedGraph<V, D>()
    PARA CADA v EN grafo.vertices() HACER
        arbol.agregarVertice(v)                         // el árbol tiene todos los vértices
    FIN PARA CADA

    MIENTRAS NO VmenosU.esVacio() HACER
        arista ← searchMinEdge(grafo, U, VmenosU)       // la más barata de U a V−U
        SI arista = nulo ENTONCES
            SALIR                                       // grafo no conexo: no hay más puentes
        FIN SI
        arbol.agregarArista(arista.source(), arista.target(), arista.dato())
        nuevo ← arista.target()                         // por construcción, target ∈ V−U
        U.agregar(nuevo)
        VmenosU.remover(nuevo)
    FIN MIENTRAS

    DEVOLVER arbol
FIN

// Devuelve la arista de menor peso con un extremo en U y el otro en V.
// La devuelve "orientada": source ∈ U y target ∈ V. Si no existe ninguna, devuelve nulo.
Edge<V, D> searchMinEdge(IUndirectedGraph<V, D> grafo, Collection<V> U, Collection<V> V)
COMIENZO
    mejor     ← nulo
    mejorPeso ← +∞
    PARA CADA u EN U HACER
        PARA CADA e EN grafo.adyacencias(grafo.construirComparable(u)) HACER
            w ← e.target()                              // el vecino de u
            SI V.contiene(w) Y e.dato().getWeight() < mejorPeso ENTONCES
                mejor     ← nuevo UndirectedEdge<V, D>(u, w, e.dato())
                mejorPeso ← e.dato().getWeight()
            FIN SI
        FIN PARA CADA
    FIN PARA CADA
    DEVOLVER mejor
FIN
